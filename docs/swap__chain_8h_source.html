94static inline VkResult create_swap_chain(Device* device, VkExtent2D window, VkSurfaceKHR surface, SwapChain* swap_chain, HeapStack* memory_arena){
95 SwapChainSupportDetails swap_chain_support = find_swap_chain_support(device->physical_device, surface, memory_arena);
96
97 VkSurfaceFormatKHR surface_format = select_swap_surface_format((VkSurfaceFormatKHR*)swap_chain_support.surface_data, swap_chain_support.surface_amount);
98 VkSurfaceFormatKHR* surface_end = (VkSurfaceFormatKHR*)swap_chain_support.surface_data;
99 surface_end+= swap_chain_support.surface_amount;
100 VkPresentModeKHR* offset = (VkPresentModeKHR*)surface_end;
101 VkPresentModeKHR present_mode = select_swap_present_mode(offset, swap_chain_support.present_amount);
102
103 VkExtent2D screen_extent = select_swap_chain_extent(swap_chain_support.surface_capabilities, window);
104
105 uint32_t image_amount = swap_chain_support.surface_capabilities.minImageCount + 1;
106
107 if(swap_chain_support.surface_capabilities.maxImageCount > 0 && image_amount > swap_chain_support.surface_capabilities.maxImageCount){
108 image_amount = swap_chain_support.surface_capabilities.maxImageCount;
109 }
110
111 VkSwapchainCreateInfoKHR create_info{};
112 create_info.sType = VK_STRUCTURE_TYPE_SWAPCHAIN_CREATE_INFO_KHR;
113 create_info.surface = surface;
114 create_info.minImageCount = image_amount;
115 create_info.imageFormat = surface_format.format;
116 create_info.imageColorSpace = surface_format.colorSpace;
117 create_info.imageExtent = screen_extent;
118 create_info.imageArrayLayers = 1;
119 create_info.imageUsage = VK_IMAGE_USAGE_COLOR_ATTACHMENT_BIT;
120
121 QueueFamilyIndices indices = find_queue_families(device->physical_device, surface, memory_arena);
122 uint32_t queue_family_indices[] = { indices.graphics_family.number, indices.present_family.number};
123
124 if (indices.graphics_family.number != indices.present_family.number) {
125 create_info.imageSharingMode = VK_SHARING_MODE_CONCURRENT;
126 create_info.queueFamilyIndexCount = 2;
127 create_info.pQueueFamilyIndices = queue_family_indices;
128 } else {
129 create_info.imageSharingMode = VK_SHARING_MODE_EXCLUSIVE;
130 create_info.queueFamilyIndexCount = 0; // Optional
131 create_info.pQueueFamilyIndices = nullptr; // Optional
132 }
133
134 create_info.preTransform = swap_chain_support.surface_capabilities.currentTransform;
135 create_info.compositeAlpha = VK_COMPOSITE_ALPHA_OPAQUE_BIT_KHR;
136 create_info.presentMode = present_mode;
137 create_info.clipped = VK_TRUE;
138
139 create_info.oldSwapchain = VK_NULL_HANDLE;
140
141 VkResult result = vkCreateSwapchainKHR(device->virtual_device, &create_info, nullptr, &swap_chain->swap_chain);
142
143 if(result != VK_SUCCESS){
144 return result;
145 //("Failed to create swap chain");
146 }
147 swap_chain->screen_extent = screen_extent;
148 swap_chain->swap_chain_image_format = surface_format.format;
149 return VK_SUCCESS;
150}

152static inline int clean_swap_chain(VkDevice virtual_device, SwapChain* swap_chain, SwapChainImages* swap_chain_images, HeapStack* heap_stack){
153 vkDeviceWaitIdle(virtual_device);
