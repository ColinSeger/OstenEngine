83 static size_t de_serialize(const char* filename, VertexArray& vertices, Uint32Array& indices, HeapStack* heap_stack){
84 Debug::profile_time_start();
85 std::ifstream file(filename, std::ios::binary | std::ios::ate);
86
87 if(!file.is_open()){
88 Debug::log((char*)"There was a issue parsing this model");
89 return heap_stack->index;

91
92 //Find out where the file ends
93 size_t file_size = file.tellg();
94 file.seekg(0);
95 size_t mem_index = arena_alloc_memory(heap_stack, file_size);
96 char* buffer = (char*)get_at_index(heap_stack, mem_index);
97
98 file.read(buffer, file_size);
99 file.close();
100 //195528 debug asset
101
102 //Figures out where the vertexes starts/stops
103 uint32_t index_start = *((uint32_t*)buffer);
104
105 //Sets where to start values of the model
106 vertices.values = (Vertex*)(buffer + sizeof(uint32_t));
107 indices.values = (uint32_t*)(buffer+ sizeof(uint32_t) + index_start);
108
109 //Sets amount the array contains
110 vertices.amount = index_start / sizeof(Vertex);
111 indices.amount = (file_size - index_start) / sizeof(uint32_t);
112
113 Debug::profile_time_end();
114 return mem_index;
