398static void update_view_buffer(Transform transform, ViewDescriptor* view_descriptor, uint8_t current_image, float aspect_ratio, float field_of_view, float range, uint8_t amount){
399
400 vec3_t forward_vector = v3_add(transform.position, v3_forward_vector(transform));
401
402 mat4_t view_matrix = m4_look_at(transform.position, forward_vector, {0, 0, 1});
403 mat4_t projection = m4_perspective_matrix(field_of_view, aspect_ratio, 0.8f, range);
404 mat4_t v_matrix = m4_mul(projection, view_matrix);
405
406 memcpy(view_descriptor->uniform_buffers_mapped[current_image], &v_matrix, sizeof(mat4_t) * amount);
407}

409static void update_view_buffer_light(Transform* transform, ViewDescriptor* view_descriptor, uint8_t current_image, float aspect_ratio, float field_of_view, float range){
410
411 mat4_t v_matrix[MAX_LIGHTS];
412 mat4_t projection = m4_perspective_matrix(field_of_view, aspect_ratio, 0.8f, range);
413 for(int light = 0; light < MAX_LIGHTS; light++){
414 vec3_t forward_vector = v3_add(transform[light].position, v3_forward_vector(transform[light]));
415
416 mat4_t view_matrix = m4_look_at(transform[light].position, forward_vector, {0, 0, 1});
417 v_matrix[light] = m4_mul(projection, view_matrix);
418 }
419
420 memcpy(view_descriptor->uniform_buffers_mapped[current_image], v_matrix, sizeof(mat4_t) * MAX_LIGHTS);
421}

593static void start_shadow_pass(VkCommandBuffer* command_buffer, const VkExtent2D viewport_extent, struct LightSources* lights, ModelData model_data, const uint8_t frame, HeapStack* heap_stack){
594 //Begining of shadow pass
595 VkClearValue clear_values[1]{};
596 clear_values[0].depthStencil = {1.0f, 0};
597
598 VkViewport viewport{};
599 viewport.x = 0.0f;
600 viewport.y = 0.0f;
601 viewport.width = ((float)viewport_extent.width);
602 viewport.height =((float)viewport_extent.height);
603 viewport.minDepth = 0.0f;
604 viewport.maxDepth = 1.0f;
605 vkCmdSetViewport(*command_buffer, 0, 1, &viewport);
606
607 VkRect2D scissor{};
608 scissor.offset = {0, 0};
609 scissor.extent = viewport_extent;
610 vkCmdSetScissor(*command_buffer, 0, 1, &scissor);
611
612 vkCmdBindPipeline(*command_buffer, VK_PIPELINE_BIND_POINT_GRAPHICS, lights->shadow_pipeline);
613 if(loaded_models.size() <= 0 || model_data.renderable_amount <= 0){
614 vkCmdEndRenderPass(*command_buffer);
615 return;
616 }
617 constexpr VkDeviceSize offsets[] = {0};
618 for(uint8_t light_index = 0; light_index < (uint8_t)lights->light_amount; light_index++){
619
620 VkRenderPassBeginInfo render_pass_info{};
621 render_pass_info.sType = VK_STRUCTURE_TYPE_RENDER_PASS_BEGIN_INFO;
622 render_pass_info.renderPass = lights->render_pass;
623 render_pass_info.framebuffer = lights->shadow_passes[light_index].framebuffer;
624 render_pass_info.renderArea.offset = {0, 0};
625 render_pass_info.renderArea.extent = viewport_extent;
626 render_pass_info.clearValueCount = sizeof(clear_values) / sizeof(clear_values[0]);
627 render_pass_info.pClearValues = clear_values;
628
629 vkCmdBeginRenderPass(*command_buffer, &render_pass_info, VK_SUBPASS_CONTENTS_INLINE);
630
631 for(uint16_t render_index = 0; render_index < model_data.renderable_amount; render_index++){
632 RenderAble* render_data = get_renderable(&model_data, render_index, heap_stack);
633
634 if(render_data == nullptr || render_data->instance_amount <= 0) continue;
635 VkDescriptorSet descriptors[] = { lights->lights_desc[light_index].descriptor_sets[frame], render_data->model_descriptor_sets[frame]};
636 uint32_t descriptor_amount = sizeof(descriptors) / sizeof(descriptors[0]);
637
638 vkCmdBindDescriptorSets(*command_buffer, VK_PIPELINE_BIND_POINT_GRAPHICS, lights->shadow_pipe_layout, 0, descriptor_amount, descriptors, 0, nullptr);
639
640 Model model = loaded_models[render_data->model_index];
641
642 if(model.index_amount > 0){
643 vkCmdBindVertexBuffers(*command_buffer, 0, 1, &model.vertex_buffer, offsets);
644
645 vkCmdBindIndexBuffer(*command_buffer, model.index_buffer, 0, VK_INDEX_TYPE_UINT32);
646
647 vkCmdDrawIndexed(*command_buffer, model.index_amount, render_data->instance_amount, 0, 0, 0);
648 }
649 }
650 vkCmdEndRenderPass(*command_buffer);
651 }
