26static inline VkDescriptorPool create_imgui_descriptor_pool(VkDevice virtual_device){
27 VkDescriptorPool imgui_pool;
28
29 VkDescriptorPoolSize pool_sizes[] = {
30 { VK_DESCRIPTOR_TYPE_COMBINED_IMAGE_SAMPLER, 100 },
31 { VK_DESCRIPTOR_TYPE_SAMPLED_IMAGE, 100 },
32 { VK_DESCRIPTOR_TYPE_UNIFORM_BUFFER, 100 },
33 { VK_DESCRIPTOR_TYPE_STORAGE_BUFFER, 100 },
34 { VK_DESCRIPTOR_TYPE_SAMPLER, 100 },
35 };
36
37 VkDescriptorPoolCreateInfo pool_info{};
38 pool_info.sType = VK_STRUCTURE_TYPE_DESCRIPTOR_POOL_CREATE_INFO;
39 pool_info.flags = VK_DESCRIPTOR_POOL_CREATE_FREE_DESCRIPTOR_SET_BIT;
40 pool_info.maxSets = 100;
41 pool_info.poolSizeCount = sizeof(pool_sizes) / sizeof(pool_sizes[0]);
42 pool_info.pPoolSizes = pool_sizes;
43
44 vkCreateDescriptorPool(virtual_device, &pool_info, nullptr, &imgui_pool);
45 return imgui_pool;
46}
